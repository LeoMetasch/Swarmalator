194    def _init_omega(self, mode: FrecMode) -> npt.NDArray[np.float64]:
195        """
196        Initialize natural frequencies based on the specified mode.
197
198        Args:
199            mode: The frequency distribution mode.
200
201        Returns:
202            Natural frequencies array.
203        """
204        match mode:
205            case FrecMode.ZERO:    return np.zeros(self.N)
206            case FrecMode.UNIFORM: return np.ones(self.N)
207            case FrecMode.BIMODAL: return np.where(np.arange(self.N) < self.N//2, 1.0, -1.0)
208            case FrecMode.RANDOM:  return np.random.uniform(-1, 1, self.N)

210    def phase_calc(self) -> tuple[npt.NDArray[np.float64] | float, npt.NDArray[np.float64] | float]:
211        """
212        Calculate phase coupling parameters Q_x and Q_theta.
213
214        Returns:
215            Tuple containing Q_x and Q_theta matrices (or scalars).
216        """
217
218        if self.freq_mode == FrecMode.ZERO:
219            return np.zeros(self.N), np.zeros(self.N)
220
221        omega_norm = np.where(self.nat_freq == 0, 1.0, np.abs(self.nat_freq))
222        omega_sign = self.nat_freq / omega_norm
223
224        s_i = omega_sign[:, None]
225        s_j = omega_sign[None, :]
226        diff_sign = np.abs(s_j - s_i)
227
228        Q_x_mat = (np.pi / 2) * diff_sign
229        Q_theta_mat = (np.pi / 4) * diff_sign
230
231        return Q_x_mat, Q_theta_mat

233    def update_velocities(self) -> tuple[npt.NDArray[np.float64], npt.NDArray[np.float64]]:
234        """
235        Update particle velocities based on phases.
236
237        Returns:
238            Tuple of x and y velocity arrays.
239        """
240        x_vel = self.nat_freq * np.cos(self.phases + np.pi / 2)
241        y_vel = self.nat_freq * np.sin(self.phases + np.pi / 2)
242        return x_vel, y_vel

244    def step(self) -> None:

251    def _step_predator(self) -> None:
252        """Handle predator dynamics (O(N), not a bottleneck)."""
253        dx_all = self.x_pos - self.pred_x
254        dy_all = self.y_pos - self.pred_y
255        dist_sq_all = dx_all**2 + dy_all**2
256
257        nearest_idx = np.argmin(dist_sq_all)
258
259        target_x = self.x_pos[nearest_idx]
260        target_y = self.y_pos[nearest_idx]
261
262        hunt_dx = target_x - self.pred_x
263        hunt_dy = target_y - self.pred_y
264        hunt_dist = np.sqrt(hunt_dx**2 + hunt_dy**2) + self.eps
265
266        self.pred_x += (hunt_dx / hunt_dist) * self.dt
267        self.pred_y += (hunt_dy / hunt_dist) * self.dt
268
269        pred_dx = self.x_pos - self.pred_x
270        pred_dy = self.y_pos - self.pred_y
271
272        d_pred2 = pred_dx**2 + pred_dy**2 + self.eps
273        d_pred = np.sqrt(d_pred2)
274
275        repulsion_mag = self.hunting_strength / d_pred2
276
277        # Apply predator repulsion to positions
278        self.x_pos += (pred_dx / d_pred) * repulsion_mag * self.dt
279        self.y_pos += (pred_dy / d_pred) * repulsion_mag * self.dt

281    def _step_numba(self) -> None:
282        """Numba-accelerated step using JIT-compiled kernel."""
283        # Prepare Q matrices for Numba kernel
284        is_scalar_Q = not self.phase_coupling
285        if is_scalar_Q:
286            # Create dummy arrays (won't be used but Numba needs consistent types)
287            Q_x = np.zeros((1, 1), dtype=np.float64)
288            Q_theta = np.zeros((1, 1), dtype=np.float64)
289        else:
290            Q_x = self.Q_x
291            Q_theta = self.Q_theta
292
293        # Call JIT-compiled kernel
294        new_phases, new_x, new_y = _step_kernel_numba(
295            self.x_pos, self.y_pos, self.phases, self.nat_freq,
296            self.vx, self.vy, self.J, self.K, self.N, self.dt, self.eps,
297            Q_x, Q_theta, is_scalar_Q
298        )
299
300        self.phases = new_phases
301        self.x_pos = new_x
302        self.y_pos = new_y
303        self.vx, self.vy = self.update_velocities()
304
305        # Handle predator separately (already O(N))
306        if self.predator:
307            self._step_predator()

309    def _step_naive(self) -> None:
310        """Original O(N²) NumPy implementation for comparison/validation."""
311        dX = self.x_pos[None, :] - self.x_pos[:, None]
312        dY = self.y_pos[None, :] - self.y_pos[:, None]
313
314        # Distances
315        dist2 = dX*dX + dY*dY
316        # exclude i==i by making the diagonal "infinite distance"
317        np.fill_diagonal(dist2, np.inf)
318        dist = np.sqrt(dist2)
319
320        dTheta = self.phases[None, :] - self.phases[:, None]
321        c = np.cos(dTheta - self.Q_x)
322        s = np.sin(dTheta - self.Q_theta)
323
324        coef = (1.0 + self.J*c) / (dist + self.eps) - 1.0 / (dist2 + self.eps)
325
326        thetadot = self.nat_freq + (self.K * (s / (dist + self.eps)).sum(axis=1)) / self.N
327
328        self.phases += thetadot * self.dt
329        self.phases = (self.phases + np.pi) % (2*np.pi) - np.pi
330
331        self.vx, self.vy = self.update_velocities()
332
333        xdot = self.vx + (dX * coef).sum(axis=1) / self.N
334        ydot = self.vy + (dY * coef).sum(axis=1) / self.N
335
336        if self.predator:
337
338            dx_all = self.x_pos - self.pred_x
339            dy_all = self.y_pos - self.pred_y
340            dist_sq_all = dx_all**2 + dy_all**2
341
342            nearest_idx = np.argmin(dist_sq_all)
343
344            target_x = self.x_pos[nearest_idx]
345            target_y = self.y_pos[nearest_idx]
346
347            hunt_dx = target_x - self.pred_x
348            hunt_dy = target_y - self.pred_y
349            hunt_dist = np.sqrt(hunt_dx**2 + hunt_dy**2) + self.eps
350
351            self.pred_x += (hunt_dx / hunt_dist) * self.dt
352            self.pred_y += (hunt_dy / hunt_dist) * self.dt
353
354            pred_dx = self.x_pos - self.pred_x
355            pred_dy = self.y_pos - self.pred_y
356
357            d_pred2 = pred_dx**2 + pred_dy**2 + self.eps
358            d_pred = np.sqrt(d_pred2)
359
360            repulsion_mag = self.hunting_strength / d_pred2
361
362            xdot += (pred_dx / d_pred) * repulsion_mag
363            ydot += (pred_dy / d_pred) * repulsion_mag
364
365        self.x_pos += xdot * self.dt
366        self.y_pos += ydot * self.dt

368    def _correlation_order_parameter(self) -> float:
369        """
370        Calculate the correlation order parameter S+-.
371
372        Returns:
373            S_plus (float): correlation order parameter S+
374            S_minus (float): correlation order parameter S-
375            Returns max(S_plus, S_minus).
376        """
377        phi = np.arctan2(self.y_pos, self.x_pos)
378        W_plus = np.exp(1j*(phi + self.phases))
379        W_minus = np.exp(1j*(phi - self.phases))
380        S_plus = np.abs(W_plus.sum()/self.N)
381        S_minus = np.abs(W_minus.sum()/self.N)
382        return max(S_plus, S_minus)

384    def _calculate_velocity_order_parameter(
385        self,
386        x_prev: npt.NDArray[np.float64],
387        y_prev: npt.NDArray[np.float64],
388        theta_prev: npt.NDArray[np.float64]
389    ) -> Tuple[float, float]:
390        """
391        Calculate the mean spatial velocity (V) and mean phase velocity (Omega) order parameters by
392        calculating the difference between the current and previous state.
393
394        Args:
395            x_prev: x-coordinates of swarmalators at previous time step
396            y_prev: y-coordinates of swarmalators at previous time step
397            theta_prev: phases of swarmalators at previous time step
398        Returns:
399            V_mean (float): mean spatial velocity
400            omega_mean (float): mean phase velocity
401        """
402        ############ Spatial velocity (V) ####################################
403        # Ensure input arrays are float/compatible
404        dx = self.x_pos - x_prev
405        dy = self.y_pos - y_prev
406        # Euclidean distance
407        V = np.sqrt(dx**2 + dy**2)
408        V_mean = float(np.mean(V))
409
410        ############ Phase velocity (Omega) ##################################
411        # Use the shortest path to calculate the phase difference - on a circle,
412        # the shortest path between two angles is not always the absolute difference
413        dtheta = self.phases - theta_prev
414        dtheta = (dtheta + np.pi) % (2 * np.pi) - np.pi
415
416        # Calculate the mean phase velocity - how fast the phases (colours) are changing in the simulation
417        # without accounting for the direction of change
418        omega_i = np.abs(dtheta) / self.dt
419
420        # Are the oscillators still oscillating?
421        omega_mean = np.mean(omega_i)
422
423        return V_mean, omega_mean

425    def _synchrony_order_parameter(self) -> float:
426        """Calculate the Kuramoto order parameter R for the current phases."""
427        return float(np.abs(np.mean(np.exp(1j * self.phases))))

429    def _circular_kmeans_labels(
430        self,
431        theta: NDArray[np.floating],
432        k: int,
433        iters: int = 30,
434        seed: int = 0,
435    ) -> NDArray[np.int64]:
436        """
437        K-means clustering on phases (angles) using the embedding theta -> (cos theta, sin theta).
438
439        This avoids problems at the branch cut (e.g. -pi and +pi represent the same angle).
440
441        Args:
442            theta: Phase angles (radians), shape (N,).
443            k: Number of clusters.
444            iters: Maximum number of k-means iterations.
445            seed: Random seed for initialization.
446
447        Returns:
448            labels: Integer cluster labels in {0..k-1}, shape (N,).
449        """
450        rng = np.random.default_rng(seed)
451        X = np.c_[np.cos(theta), np.sin(theta)]  # (N,2) points on the unit circle
452
453        # Initialize: pick k random points as centers
454        cent = X[rng.choice(X.shape[0], size=k, replace=False)].copy()
455
456        for _ in range(iters):
457            # Assignment: squared distances to centers (N,k)
458            d2 = ((X[:, None, :] - cent[None, :, :]) ** 2).sum(axis=2)
459            lab = np.argmin(d2, axis=1)
460
461            # Update: mean direction per cluster, renormalized to unit length
462            new_cent = cent.copy()
463            for j in range(k):
464                pts = X[lab == j]
465                if pts.shape[0] == 0:
466                    new_cent[j] = X[rng.integers(0, X.shape[0])]
467                else:
468                    v = pts.mean(axis=0)
469                    n = np.linalg.norm(v)
470                    new_cent[j] = v / (n + 1e-12)
471
472            if np.allclose(new_cent, cent):
473                break
474            cent = new_cent
475
476        return lab.astype(np.int64)

478    def _cluster_separation_score(
479        self,
480        x: NDArray[np.floating],
481        y: NDArray[np.floating],
482        labels: NDArray[np.integer],
483    ) -> float:
484        """
485        Measure spatial separation of label-defined groups.
486
487        Score = (mean inter-centroid distance) / (mean within-cluster spread).
488        Higher values indicate more clearly separated spatial lobes.
489
490        Args:
491            x: x-coordinates, shape (N,).
492            y: y-coordinates, shape (N,).
493            labels: Integer labels defining groups, shape (N,).
494
495        Returns:
496            separation: Dimensionless separation score (>= 0).
497        """
498        k = int(labels.max()) + 1
499        if k < 2:
500            return 0.0
501
502        pts = np.c_[np.asarray(x, dtype=float), np.asarray(y, dtype=float)]
503        cents = np.array([pts[labels == j].mean(axis=0) for j in range(k)])
504
505        # Within-cluster spread (mean distance to centroid)
506        spreads = []
507        for j in range(k):
508            pj = pts[labels == j]
509            if pj.shape[0] < 2:
510                continue
511            spreads.append(np.sqrt(((pj - cents[j]) ** 2).sum(axis=1)).mean())
512        within = float(np.mean(spreads)) if spreads else 0.0
513
514        # Between-centroid distances (mean pairwise distance)
515        d = []
516        for a in range(k):
517            for b in range(a + 1, k):
518                d.append(np.linalg.norm(cents[a] - cents[b]))
519        between = float(np.mean(d)) if d else 0.0
520
521        return between / (within + 1e-12)

523    def _phase_compactness(self, labels: NDArray[np.integer]) -> float:
524        """
525        Compute weighted within-cluster circular coherence of phases.
526
527        For each cluster j:
528            r_j = |mean(exp(i*theta_i))| over i in cluster j
529        The returned compactness is the size-weighted average of r_j across clusters.
530
531        Args:
532            labels: Integer cluster labels, shape (N,).
533
534        Returns:
535            compactness: Value in [0, 1]; higher means tighter phase groups within clusters.
536        """
537        k = int(labels.max()) + 1
538        N = int(labels.size)
539        if k < 1 or N == 0:
540            return 0.0
541
542        z = np.exp(1j * self.phases)
543        comp = 0.0
544        for j in range(k):
545            idx = labels == j
546            nj = int(idx.sum())
547            if nj == 0:
548                continue
549            rj = float(np.abs(z[idx].mean()))
550            comp += (nj / N) * rj
551        return float(comp)

553    def _spatial_anisotropy(self, labels: NDArray[np.integer]) -> float:
554        """
555        Compute weighted mean spatial anisotropy of clusters.
556
557        For each cluster, compute covariance eigenvalues (lmax, lmin) of (x, y) and define:
558            anisotropy = sqrt((lmax + tiny) / (lmin + tiny))
559
560        Blobs -> ~1, arcs/filaments -> large.
561
562        Args:
563            labels: Integer cluster labels, shape (N,).
564
565        Returns:
566            anisotropy: Weighted mean anisotropy across clusters.
567        """
568        k = int(labels.max()) + 1
569        N = int(labels.size)
570        if k < 1 or N == 0:
571            return 0.0
572
573        pts = np.c_[self.x_pos, self.y_pos]
574        anisos = []
575        weights = []
576        for j in range(k):
577            idx = labels == j
578            nj = int(idx.sum())
579            if nj < 3:
580                continue
581            pj = pts[idx]
582            cov = np.cov(pj.T)
583            try:
584                w = np.linalg.eigvalsh(cov)
585            except np.linalg.LinAlgError:
586                continue
587            lmin = float(np.min(w))
588            lmax = float(np.max(w))
589            aniso = float(np.sqrt((lmax + 1e-12) / (lmin + 1e-12)))
590            anisos.append(aniso)
591            weights.append(nj)
592
593        if not anisos:
594            return 0.0
595
596        weights_arr = np.array(weights, dtype=float)
597        weights_arr /= weights_arr.sum()
598        return float(np.dot(weights_arr, np.array(anisos, dtype=float)))

600    def splinter_diagnostics(
601        self,
602        k_min: int = 2,
603        k_max: int = 12,
604        min_frac: float = 0.05,
605        seed: int = 0,
606        k_penalty: float = 1.5,
607    ) -> Tuple[int, float, float, float, Optional[NDArray[np.int64]]]:
608        """
609        Diagnose whether phases form discrete groups that are also spatially separated.
610
611        This combines:
612          - spatial separation of phase-defined groups, and
613          - within-cluster phase compactness (circular coherence).
614
615        To reduce false positives (e.g. slicing a smooth phase wave into many groups), the
616        score penalizes large k and highly anisotropic (arc-like) spatial clusters.
617
618        Args:
619            k_min: Minimum number of phase clusters to evaluate.
620            k_max: Maximum number of phase clusters to evaluate.
621            min_frac: Minimum fraction of points required in each cluster (tiny clusters ignored).
622            seed: Random seed for clustering initialization.
623            k_penalty: Exponent controlling the penalty for large k in the score.
624
625        Returns:
626            best_k: Selected number of phase clusters.
627            best_sep: Spatial separation score (higher => clearer lobes).
628            best_comp: Phase compactness in [0, 1] (higher => tighter phase groups).
629            best_aniso: Spatial anisotropy (higher => more arc/filament-like).
630            labels: Cluster labels for best_k, or None if no valid clustering was found.
631        """
632        best = {"k": 1, "sep": 0.0, "comp": 0.0, "aniso": 0.0, "labels": None}
633        theta = self.phases
634
635        for k in range(int(k_min), int(k_max) + 1):
636            labels = self._circular_kmeans_labels(theta, k, seed=seed)
637
638            counts = np.bincount(labels, minlength=k)
639            if counts.min() < min_frac * theta.size:
640                continue
641
642            sep = self._cluster_separation_score(self.x_pos, self.y_pos, labels)
643            comp = self._phase_compactness(labels)
644            aniso = self._spatial_anisotropy(labels)
645
646            score = (sep * comp) / (k ** k_penalty) / (1.0 + aniso)
647            best_score = (best["sep"] * best["comp"]) / (max(best["k"], 1) ** k_penalty) / (1.0 + best["aniso"])
648            if score > best_score:
649                best = {"k": k, "sep": float(sep), "comp": float(comp), "aniso": float(aniso), "labels": labels}
650
651        return (
652            int(best["k"]),
653            float(best["sep"]),
654            float(best["comp"]),
655            float(best["aniso"]),
656            best["labels"].astype(np.int64) if best["labels"] is not None else None,
657        )

659    def stability_analysis(self) -> Tuple[str, float, float, float, int, float, float, float]:
660        """
661        Analyze the current system state using order parameters and diagnostics.
662
663        Returns:
664            state: Human-readable state label.
665            S_parameter: Correlation order parameter S.
666            V_parameter: Mean spatial velocity magnitude per step.
667            omega_parameter: Mean phase angular velocity magnitude (rad/time).
668            best_k: Best number of phase clusters (diagnostics).
669            best_sep: Spatial separation score of phase clusters.
670            best_comp: Within-cluster phase compactness (circular coherence).
671            best_aniso: Spatial anisotropy of phase clusters.
672        """
673        x_prev = self.x_pos.copy()
674        y_prev = self.y_pos.copy()
675        theta_prev = self.phases.copy()
676
677        self.step()
678
679        S_parameter = self._correlation_order_parameter()
680        V_parameter, omega_parameter = self._calculate_velocity_order_parameter(x_prev, y_prev, theta_prev)
681        R_parameter = self._synchrony_order_parameter()
682
683        best_k, best_sep, best_comp, best_aniso, _ = None, None, None, None, None
684
685        # 1. Check for Active Phase Wave (Specific dynamic state)
686        if S_parameter > 0.5 and V_parameter >= 0.001 and omega_parameter >= 0.01:
687            state = "Active Phase Wave"
688
689        # 2. Check for Static Async (Disordered)
690        elif S_parameter <= 0.5:
691            # Paper calls low S "Static Async".
692            # Note: We check R as well to distinguish Sync if S is low but R is high (unlikely but robust)
693            state = "Static Sync" if R_parameter > 0.9 else "Static Async"
694

210    def phase_calc(self) -> tuple[npt.NDArray[np.float64] | float, npt.NDArray[np.float64] | float]:
211        """
212        Calculate phase coupling parameters Q_x and Q_theta.
213
214        Returns:
215            Tuple containing Q_x and Q_theta matrices (or scalars).
216        """
217
218        if self.freq_mode == FrecMode.ZERO:
219            return np.zeros(self.N), np.zeros(self.N)
220
221        omega_norm = np.where(self.nat_freq == 0, 1.0, np.abs(self.nat_freq))
222        omega_sign = self.nat_freq / omega_norm
223
224        s_i = omega_sign[:, None]
225        s_j = omega_sign[None, :]

227
228        Q_x_mat = (np.pi / 2) * diff_sign
229        Q_theta_mat = (np.pi / 4) * diff_sign
230
231        return Q_x_mat, Q_theta_mat

def splinter_diagnostics( self, k_min: int = 2, k_max: int = 12, min_frac: float = 0.05, seed: int = 0, k_penalty: float = 1.5) -> Tuple[int, float, float, float, Optional[numpy.ndarray[tuple[int, ...], numpy.dtype[numpy.int64]]]]: 
600    def splinter_diagnostics(
601        self,
602        k_min: int = 2,
603        k_max: int = 12,
604        min_frac: float = 0.05,
605        seed: int = 0,
606        k_penalty: float = 1.5,
607    ) -> Tuple[int, float, float, float, Optional[NDArray[np.int64]]]:
608        """
609        Diagnose whether phases form discrete groups that are also spatially separated.
610
611        This combines:
612          - spatial separation of phase-defined groups, and
613          - within-cluster phase compactness (circular coherence).
614
615        To reduce false positives (e.g. slicing a smooth phase wave into many groups), the
616        score penalizes large k and highly anisotropic (arc-like) spatial clusters.
617
618        Args:
619            k_min: Minimum number of phase clusters to evaluate.
620            k_max: Maximum number of phase clusters to evaluate.
621            min_frac: Minimum fraction of points required in each cluster (tiny clusters ignored).
622            seed: Random seed for clustering initialization.
623            k_penalty: Exponent controlling the penalty for large k in the score.
624
625        Returns:
626            best_k: Selected number of phase clusters.
627            best_sep: Spatial separation score (higher => clearer lobes).
628            best_comp: Phase compactness in [0, 1] (higher => tighter phase groups).
629            best_aniso: Spatial anisotropy (higher => more arc/filament-like).
630            labels: Cluster labels for best_k, or None if no valid clustering was found.
631        """
632        best = {"k": 1, "sep": 0.0, "comp": 0.0, "aniso": 0.0, "labels": None}
633        theta = self.phases
634
635        for k in range(int(k_min), int(k_max) + 1):
636            labels = self._circular_kmeans_labels(theta, k, seed=seed)
637
638            counts = np.bincount(labels, minlength=k)
639            if counts.min() < min_frac * theta.size:
640                continue
641
642            sep = self._cluster_separation_score(self.x_pos, self.y_pos, labels)
643            comp = self._phase_compactness(labels)
644            aniso = self._spatial_anisotropy(labels)
645
646            score = (sep * comp) / (k ** k_penalty) / (1.0 + aniso)
647            best_score = (best["sep"] * best["comp"]) / (max(best["k"], 1) ** k_penalty) / (1.0 + best["aniso"])
648            if score > best_score:
649                best = {"k": k, "sep": float(sep), "comp": float(comp), "aniso": float(aniso), "labels": labels}
650
651        return (
652            int(best["k"]),
653            float(best["sep"]),
654            float(best["comp"]),
655            float(best["aniso"]),
656            best["labels"].astype(np.int64) if best["labels"] is not None else None,
657        )
